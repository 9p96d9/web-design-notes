script.js

  • 関数名「calculateSum」を作った場合
const result = () => {
  // 入力フィールドから値を取得
  const numberInput = document.querySelector('#numberInput').value;
  // 入力された数値の整数化（切り捨て）
  const num = parseInt(numberInput);

  // 入力が有効な整数かどうかをチェック
  if (isNaN(num) || num < 0) {
      alert('有効な正の整数を入力してください。');
      return;
  }

  // 合計を計算
  let sum = 0;
  for (let i = 1; i <= num; i++) {
      sum += i;
  }

  // 結果を表示
  document.querySelector('#result').textContent = `0から${num}までの合計は: ${sum}`;
};

// ボタンにクリックイベントを追加
document.querySelector('#calculateSum').addEventListener('click', result);
  • 無名関数の場合
document.querySelector('#calculateSum').addEventListener('click', () => {
  // 入力フィールドから値を取得
  const numberInput = document.querySelector('#numberInput').value.trim();
  const num = parseInt(numberInput, 10);

  // 入力が有効な整数かどうかをチェック
  if (isNaN(num) || num < 0) {
    alert('有効な正の整数を入力してください。');
    return;
  }

  // 合計を計算
  let sum = 0;
  for (let i = 1; i <= num; i++) {
    sum += i;
  }

  // 結果を表示
  document.querySelector('#result').textContent = `0から${num}までの合計は: ${sum}`;
});
