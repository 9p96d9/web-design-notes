script.js

$(function(){

  $('#slider').slick({
    autoplay: true,
    autoplaySpeed: 3000,
    speed: 2000,
    pose: 4000,
    arrows: true,
    dots: true,
    slidesToShow: 1,
    centerMode: true, 
    centerPadding: '18%',
    prevArrow:'<div class="prev"></div>',
    nextArrow:'<div class="next"></div>',
    responsive: [
      {
        breakpoint: 768,
        settings: {
          centerMode: false
        }
      }
    ]
  });

});
