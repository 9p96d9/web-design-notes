index.html

<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>お問い合わせ</title>
<link rel="stylesheet" href="css/style.css">
</head>
<body>
<div class="container">
  <section id="formSec" class="form">
    <h2>お問い合わせ</h2>
    <form action="" method="post">
    <table>
      <tr>
      <th>お名前</th>
      <td><input type="text" name="" required autofocus placeholder="お名前"></td>
      </tr>
      <tr>
      <th>メールアドレス</th>
      <td><input type="email" name="" required placeholder="メールアドレス"></td>
      </tr>
      <tr>
      <th class="msg">メッセージ</th>
      <td><textarea name="" cols="30" rows="10" placeholder="メッセージ"></textarea></td>
      </tr>
    </table>
    <p class="btn"><input type="submit" value="送信"></p>
    </form>
  </section>
</div><!-- /.container -->
</body>
</html>

style.css

  • ボタンの大きさは「44px」以上にする
  • マルチデバイス対応は、「min()」で設定
@charset "UTF-8";

/* ---------------------------------------------
  reset
--------------------------------------------- */
*, *::before, *::after {
  margin: 0 ;
  padding: 0;
  box-sizing: border-box;
}
html {
  font-size: 62.5%;
}

/* ---------------------------------------------
  body
--------------------------------------------- */
body {
  background-color: #fff;
  color: #333;
  font-size: 1.6rem;
  font-family: sans-serif;
  line-height: 1.0;
}


/* ---------------------------------------------
  layout
--------------------------------------------- */
.container {
    width: min(86%, 1024px);
    margin: 0 auto;
}


/* ---------------------------------------------
  .form
--------------------------------------------- */
.form {
  max-width: 640px;
  margin: 0 auto;
  padding: 60px 0;
}
.form h2 {
  margin-bottom: 30px;
  font-size: 2.8rem;
  text-align: center;
}
.form table {
  width: 100%;
  text-align: left;
}
  .form th, .form td {
    display: block;
  }
  .form th {
    width: fit-content;
    margin-bottom: 6px;
    position: relative;
  }
    .form th::after {
      content: '必須';
      position: absolute;
      top: -2px;
      right: -4.0rem;
      padding: 4px 6px;
      background: #f00;
      color: #fff;
      font-size: 0.6rem;
    }
    .form th.msg::after {
      display: none;
    }
  .form td {
    width: 100%;
    margin-bottom: 20px;
  }
  .form td label {
    padding-right: 20px;
  }
.form input[type="text"], .form input[type="email"] {
  padding: 7px 9px;
  width: 100%;
}
.form textarea {
  padding: 7px 9px;
  width: 100%;
  line-height: 1.7;
}

/* ---------- 送信ボタン ---------- */
  .btn {
    text-align: center;
  }
  .btn input[type="submit"] {
    padding: 20px 60px;
    background-color: #0163ba;
    color: #fff;
    font-size: 2.0rem;
    border: none;
    border-radius: 10px;
  }
  .btn input:hover {
    background-color: #ea4504;
  }
