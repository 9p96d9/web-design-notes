style.css

@charset "UTF-8";

/* ---------------------------------------
  reset
--------------------------------------- */
*, *::before, *::after {
  margin: 0;
  padding: 0;
  box-sizing: border-box;
}
ul, li {
  list-style: none;
}
a {
  color: inherit;
  text-decoration: none;
}
img {
  max-width: 100%;
  vertical-align: bottom;
}
html {
  font-size: 100%;
}


/* ---------------------------------------
  body
--------------------------------------- */
body {
  background-color: #666;
  color: #fff;
  font-size: 1.0rem;
  font-family: Arial, sans-serif;
  line-height: 1.0;
}
#top {
  width: 100%;
  height: 100vh;
  background-image: linear-gradient(25deg, 
  rgba(178, 5, 5, 0.6), 
  rgba(4, 160, 4, 0.6)), 
  url(../img/grass.webp);
  background-repeat: no-repeat;
  background-position: center center;
  background-size: cover;
}


/* ---------------------------------------
  layout
--------------------------------------- */
.container {
  display: flex;
  align-items: center;
}

@media screen and (max-width: 767px) {
  .container {
    display: block;
  }
}


/* ---------------------------------------
  header
--------------------------------------- */
.header {
  position: fixed;
  top: 0;
  width: 100%;
}
  .header > .container {
    width: min(92%, 960px);

    text-align: center;
  }

      font-family: 'Caveat', cursive;
    }
@media screen and (max-width: 767px) {
  .header h1 {
    padding-top: 2vh;
    margin: 0 auto 1.5vh 0;
    font-size: 2.4rem;
  }
}


/* ---------------------------------------
  nav
--------------------------------------- */
.nav_button {
  display: none;
}
.gnav > ul {
    display: flex;
    justify-content: center;
    gap: 40px;
  }
    .gnav ul li a {
      display: block;
      padding: 8px 10px;
      font-size: 1.25rem;

      white-space: nowrap;
    }

        background-color: #444;
        color: #fff;
      }
@media screen and (max-width: 767px) {
  .nav_button {
    position: fixed;
    right: 0;
    top: 0;
    width: 50px;

    transition: all 0.2s ease-in-out;
  }

  .open .nav_button span:nth-child(1) {
    transform: translate(0, 10px) rotate(45deg);
    background: #fff;
  }
  .open .nav_button span:nth-child(2) {
    transform: scaleX(0);
    opacity: 0;
  }
  .open .nav_button span:nth-child(3) {

        right: 0;
      }
