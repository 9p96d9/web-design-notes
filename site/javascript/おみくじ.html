おみくじ

おみくじの結果をブラウザに表示
  • 「大吉」のときに「文字色：赤、文字サイズ：大きく」

記述例（ES6）

index.html

<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>おみくじの結果を表示</title>
<link rel="stylesheet" href="css/style.css">
<script src="js/script.js" defer></script>
</head>
<body>

<!-- header -->
<header class="header">
  <h1>JavaScript<br>おみくじの結果を表示</h1>
  <p class="btn"><a href="../../index.html">トップへ戻る</a></p>
</header>
<!-- /header -->

<!-- main -->
<main class="main">

  <div class="container">
    <p class="question">ボタンを押してください。
      おみくじの結果が表示されます。</p>
    <button id="btn" onclick="result()">結果を表示</button>
    <h2 id="ans">ここに結果を表示</h2>
  </div><!-- /.container -->

</main>
<!-- /main -->

<!-- footer -->
<footer class="footer">
  <p id="update">制作日：xxxx年x月x日</p>
</footer>
<!-- /footer -->

</body>
</html>

script.js

// /ボタン要素と表示要素を取得
const btn = document.querySelector('#btn');
const ans = document.querySelector('#ans');

// おみくじの値を配列で設定
const luck = ['大吉', '吉', '中吉', '小吉', '凶'];

// クリックイベントが発生したとき処理される関数
btn.addEventListener('click', () => {
  // 配列の値をランダムに取得
  const rnd = Math.floor(Math.random() * luck.length);

  ans.textContent = luck[rnd];
  if (rnd == 0) {
    ans.classList.add('good');
  } else {
    ans.classList.remove('good');
  }
});

script.js

function result() {

  // おみくじの値を配列で設定
  const omikuji = ['大吉', '吉', '中吉', '小吉', '凶'];
  const rnd = Math.floor(Math.random() * omikuji.length);

  // 表示要素を取得
  const element = document.querySelector('#ans');
  element.textContent = omikuji[rnd];

  // 配列[0]のときclass名を追加
  if (rnd == 0) {
    element.classList.add('good');
  } else {
    element.classList.remove('good');
  }

}
