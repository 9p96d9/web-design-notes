script.js

const button = document.querySelector('#btn');
const ansElement = document.querySelector('#ans');

// 配列を利用してサイコロの目をランダムに表示
const result = () => {
  const dice = ['one', 'two', 'three', 'four', 'five', 'six'];

  // 乱数を整数化して配列内のサイコロの目の位置を取得
  const rnd = Math.floor(Math.random() * dice.length);

  // 配列内の位置の数字を画像のファイル名に反映させる
  ansElement.innerHTML = `<img src="img/${dice[rnd]}.png" alt="${dice[rnd]}">`;
};

// ボタンAにイベントリスナーを追加
button.addEventListener('click', result);
