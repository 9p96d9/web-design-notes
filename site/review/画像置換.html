script.js

$(function(){

  /* =================================================
  画像切り替え
  ================================================= */
  // サムネイルの画像をクリックしたときの処理
  $('#thumbnail img').on('click', function(){

    // クリックされた画像のパスを取得
    let img_src = $(this).attr("src");

    // メイン画像と同じサムネイルがクリックされたときは画像切り替えを実行しない
    // （メイン画像のsrcとクリックされた画像のsrcが異なる場合のみ実行）
    if($('#mainvisual img').attr("src") != img_src) {
      $(".main_visual img").attr({src:img_src});
    }
  });
  
});
