アコーディオンパネルの開閉
  • この記述より前に、jQuery本体が読み込まれていることが必須
  • 記述は、jQueryのイベントで記述
$(function(){
  // アコーディオンパネルの開閉
  $('.accordion > dd').hide();
  $('.accordion > dt').on('click', function(){
    if ($(this).next().css('display') == 'none') {
      $('.accordion > dd').slideUp('slow');
      $(this).next().slideDown('slow');
    }
  });
});
  • この時点で、開閉の確認

クリックされた位置（dt）にクラス名を追加（または削除）
$(function(){
  // アコーディオンパネルの開閉
  $('.accordion > dd').hide();
  $('.accordion > dt').on('click', function(){
    if ($(this).next().css('display') == 'none') {
      // クラス名を削除
      $('.accordion > dt').removeClass('open');
      $('.accordion > dd').slideUp('slow');
      // クラス名を追加
      $(this).addClass('open');
      $(this).next().slideDown('slow');
    }
  });
});

id名「now」の処理
  • この位置に「今日の日付と曜日」を表示
// 今日の日付と曜日を取得して表示
const today = new Date();
const year = today.getFullYear();
const month = today.getMonth()+1;
const date = today.getDate();
const day = today.getDay();
const week = ['日','月','火','水','木','金','土'];
const nowelement = document.querySelector('#now');
nowelement.textContent = `今日は、${year}年${month}月${date}日、${week[day]}曜日です。`;
id名「fortune」の処理
  • この位置に「おみくじ」の結果を表示
// 今日の運勢をランダムに表示
const luck = ['大吉','吉','中吉','小吉','凶'];
const rnd = Math.floor(Math.random()*luck.length);
const luckelement = document.querySelector('#fortune');
luckelement.textContent = `${luck[rnd]}`;
if(rnd === 0) {
  luckelement.classList.add('good');
} else {
  luckelement.classList.remove('good');
}
