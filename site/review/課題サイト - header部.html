header部内の構造とレイアウト
  • レイアウト用の「div」タグを追加
<!-- header -->
<header class="header">
  <div class="container">
    <h1>課題サイト</h1>
    <nav class="gnav">
      <ul>
        <li><a href="#">Website</a></li>
        <li><a href="#">Banner</a></li>
        <li><a href="#">JavaScript</a></li>
      </ul>
    </nav>
  </div><!-- /.container -->
</header>
<!-- /header -->
CSSで修飾
@charset "UTF-8";

/* ------------------------------------
  reset（空間：要素間の空きをリセット）
  レイアウトに影響が出る空きをリセット
------------------------------------ */
* {
  margin: 0;
  padding: 0;
  box-sizing: border-box;
}
ul {
  list-style: none;
}
a {
  color: inherit;
  text-decoration: none;
}
img {
  max-width: 100%;
  vertical-align: bottom;
}


/* ------------------------------------
  body（ブラウザに表示する文字を初期化）
------------------------------------ */
body {
  background-color: #fff;
  color: #333;
  font-size: 16px;  /* ブラウザの初期値 */
  font-family: sans-serif;
  line-height: 1.0;
}


/* ------------------------------------
  layout（レイアウト用共通要素：div）
------------------------------------ */
.container {
  width: 90%;  /* スマートフォン幅 */
  max-width: 960px;  /* PC幅(可変後の最大値) */
  margin: 0 auto;  /* ブラウザ幅の中央配置 */
}


/* ------------------------------------
  header
------------------------------------ */
.header {
  width: 100%;
  height: 60vh;  /* vh(view height) */
  padding: 30px 0;
  background-image: url(../img/header_bg2.jpg);
  background-repeat: no-repeat;
  background-position: left bottom;
  background-size: cover;
}
  .header > .container {
    display: flex;
    align-items: center;
  }
  h1 {
    margin-right: auto;  /* navまでのスペースを自動処理 */
    color: #fff;
  }


/* ------------------------------------
  nav
------------------------------------ */
.gnav > ul {
  display: flex;
}
  .gnav li {
    font-size: 18px;
  }
  .gnav li > a {
    display: block;  /* ボタンの面を作る */
    padding: 14px 30px;  /* 文字サイズ＋paddingで44px以上にする */
    color: #fff;
  }
  .gnav li > a:hover {
    background-color: #1a486a;
  }
